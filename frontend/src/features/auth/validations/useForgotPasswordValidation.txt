import { ForgotPasswordDTO } from "@/domain/auth/dto/ForgotPasswordDTO";
import { verifyZodImplementsInterface } from "@/libs/helpers/validations/interface/verifyZodImplementsInterface";
import { z } from "zod";

export const useForgotPasswordValidation = () => {
  const forgotPasswordValidation = z.object({
    email: z.string().min(3, { message: "Campo obrigatório" }).email({
      message: "Tente utilizar um e-mail válido!",
    }),
  });
  verifyZodImplementsInterface<ForgotPasswordDTO>(
    {} as z.infer<typeof forgotPasswordValidation>
  );
  return { forgotPasswordValidation };
};
