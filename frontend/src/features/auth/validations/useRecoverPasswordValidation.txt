import { z } from "zod";
import { verifyZodImplementsInterface } from "@/libs/helpers/validations/interface/verifyZodImplementsInterface";
import { regexPasswordComplete } from "@/libs/helpers/validations/password/regexPassword";
import { RecoverPasswordDTO } from "@/domain/auth/dto/RecoverPasswordDTO";

export const useRecoverPasswordValidation = () => {
  const recoverPasswordValidation = z
    .object({
      password: z.string().min(3, { message: "Campo obrigatório" }),
      passwordConfirmation: z.string().min(3, { message: "Campo obrigatório" }),
    })
    .refine(
      ({ password, passwordConfirmation }) => password === passwordConfirmation,
      {
        message: "As senhas não coincidem",
        path: ["passwordConfirmation"],
      }
    )
    .refine(({ password }) => regexPasswordComplete.test(password), {
      message: "A senha é fraca",
      path: ["password"],
    });
  verifyZodImplementsInterface<RecoverPasswordDTO>(
    {} as z.infer<typeof recoverPasswordValidation>
  );
  return { recoverPasswordValidation };
};
